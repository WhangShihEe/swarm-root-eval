# Pinned toolchain for the emulator harness.
# Values are the SDK package revisions actually installed & verified on this host
# (Apple Silicon M4 Max, arm64). `bin/doctor` asserts the live toolchain matches.

# Android SDK
SDK_PLATFORM_API=37.1
SYSTEM_IMAGE="system-images;android-35;google_apis;arm64-v8a"
SYSTEM_IMAGE_REVISION=9          # rootable google_apis image; the playstore variant is NOT rootable
EMULATOR_REVISION=37.1.11
PLATFORM_TOOLS_REVISION=37.0.1   # provides adb

# cmdline-tools (sdkmanager + avdmanager); needs JDK >= 17.
# Treated by doctor as a MINIMUM floor, not an exact pin: the revision must be new
# enough to parse the image's SDK-XML v4. 12.0 silently writes target=android-0 into
# the AVD config -> black-screen boot (see RUNBOOK / ticket 02). 23.0 ships as the
# 'latest-2' cmdline-tools package; env.sh prefers the newest installed revision.
CMDLINE_TOOLS_MIN_REVISION=23.0

# yq — the harness's YAML dependency (mikefarah yq, v4 API). Ambient dev tool, not an
# SDK package; pinned here as a floor so it is not assumed. Install: brew install yq.
YQ_VERSION=4.44.3

# Frida — client (host) and frida-server (device) MUST be the same version
FRIDA_VERSION=17.17.0
FRIDA_SERVER_ARCH=android-arm64
